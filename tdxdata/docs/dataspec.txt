根据您的需求，我将为您设计一套兼顾实时行情与历史数据的标准化CSV数据格式方案。该方案综合了知识库中多个数据源（Tushare、AKShare、迅投QMT、通达信TdxQuant）的字段定义，旨在提供统一、完整且易于分析的数据结构。

一、 设计原则
完整性：覆盖行情分析所需的核心字段（价格、成交量、成交额、财务指标、盘口等）。
一致性：历史与实时数据的核心字段（如open, high, low, close, volume）定义保持一致，便于跨周期分析。
可扩展性：通过data_type字段区分数据类别，支持未来增加新数据（如逐笔成交、财务数据）。
实用性：字段命名清晰，采用英文小写蛇形命名，便于程序处理和数据库存储。

四、 数据获取与生成代码示例（整合方案）
以下代码展示了如何根据上述格式，使用通达信TdxQuant获取数据并生成标准CSV。

import sys
import os
import pandas as pd
from datetime import datetime
from pathlib import Path

# 1. 配置通达信路径
TDX_INSTALL_PATH = r"D:\lwj\new_tdx"  # 请修改为实际路径
sys.path.insert(0, os.path.join(TDX_INSTALL_PATH, "PYPlugins", "user"))
from tqcenter import tq

# 2. 数据转换函数 (与历史方案一致，确保输出标准字段)
def raw_data_to_standard_df(raw_data, period='1d'):
    """将TdxQuant原始数据转换为标准历史K线格式DataFrame"""
    if not raw_data:
        return pd.DataFrame()
    
    fields = list(raw_data.keys())
    stock_codes = list(raw_data[fields[0]].keys()) if fields else []
    
    df_list = []
    for code in stock_codes:
        stock_data = {}
        for field in fields:
            series = raw_data[field].get(code)
            stock_data[field.lower()] = series.values if series is not None else None
        
        # 构建DataFrame
        temp_df = pd.DataFrame(stock_data)
        temp_df['symbol'] = code
        if series is not None and hasattr(series, 'index'):
            temp_df['trade_date'] = series.index.date  # 提取日期部分
            if period.endswith('m'):  # 如果是分钟数据，提取时间
                temp_df['time'] = series.index.time
        
        df_list.append(temp_df)
    
    if df_list:
        final_df = pd.concat(df_list, ignore_index=True)
        # 重命名映射到标准字段名 (根据TdxQuant实际返回字段调整)
        column_mapping = {
            'open': 'open',
            'high': 'high',
            'low': 'low',
            'close': 'close',
            'volume': 'volume',
            'amount': 'amount'
        }
        final_df.rename(columns=column_mapping, inplace=True)
        # 添加周期和来源标识
        final_df['period'] = period
        final_df['data_source'] = 'tdx'
        return final_df
    return pd.DataFrame()

# 3. 获取历史K线并保存为标准格式
def fetch_history_kline(symbol_list, start_date, end_date, period='1d', output_dir='./stock_data/raw/kline'):
    """获取历史K线并保存为标准CSV"""
    try:
        tq.initialize(__file__)
        tq.refresh_cache()
        
        raw_data = tq.get_market_data(
            stock_list=symbol_list,
            start_time=start_date,
            end_time=end_date,
            period=period,
            dividend_type='front',
            field_list=['Open', 'High', 'Low', 'Close', 'Volume', 'Amount'],
            fill_data=True
        )
        
        df_standard = raw_data_to_standard_df(raw_data, period)
        
        if not df_standard.empty:
            # 按股票和月份组织文件
            for symbol in df_standard['symbol'].unique():
                df_symbol = df_standard[df_standard['symbol'] == symbol].copy()
                df_symbol['year_month'] = df_symbol['trade_date'].astype(str).str[:7].str.replace('-', '')
                for ym, group in df_symbol.groupby('year_month'):
                    filename = f"{symbol}_{period}_{ym}.csv"
                    filepath = Path(output_dir) / filename
                    filepath.parent.mkdir(parents=True, exist_ok=True)
                    # 按标准字段顺序保存
                    std_columns = ['symbol', 'trade_date', 'time', 'open', 'high', 'low', 'close', 
                                   'volume', 'amount', 'period', 'data_source']
                    group = group[[col for col in std_columns if col in group.columns]]
                    group.to_csv(filepath, index=False, encoding='utf-8-sig')
                    print(f"已保存: {filepath}")
    finally:
        tq.close()

# 4. 获取实时快照并保存为标准格式 (模拟，需根据TdxQuant的snapshot接口调整字段)
def fetch_realtime_snapshot(symbol_list, output_dir='./stock_data/raw/snapshot'):
    """获取实时快照并保存为标准CSV (示例框架)"""
    try:
        tq.initialize(__file__)
        snapshot_list = []
        for code in symbol_list:
            # 注意：此处需要根据TdxQuant的 get_market_snapshot 实际返回结构调整
            snap = tq.get_market_snapshot(stock_code=code)
            if snap and snap.get('ErrorId') == '0':
                # 将快照数据映射到标准格式
                standard_snap = {
                    'symbol': code,
                    'update_time': datetime.now().strftime('%Y-%m-%d %H:%M:%S'),
                    'last_price': snap.get('LastPrice', 0),
                    'pre_close': snap.get('PreClosePrice', 0),
                    'open': snap.get('OpenPrice', 0),
                    'high': snap.get('HighPrice', 0),
                    'low': snap.get('LowPrice', 0),
                    'volume': snap.get('Volume', 0),
                    'amount': snap.get('Amount', 0),
                    'bid_price_1': snap.get('BidPrice1', 0),
                    'bid_volume_1': snap.get('BidVolume1', 0),
                    'ask_price_1': snap.get('AskPrice1', 0),
                    'ask_volume_1': snap.get('AskVolume1', 0),
                    'data_source': 'tdx'
                }
                snapshot_list.append(standard_snap)
        
        if snapshot_list:
            df_snap = pd.DataFrame(snapshot_list)
            today_str = datetime.now().strftime('%Y-%m-%d')
            filepath = Path(output_dir) / f"snapshot_{today_str}.csv"
            filepath.parent.mkdir(parents=True, exist_ok=True)
            df_snap.to_csv(filepath, index=False, encoding='utf-8-sig')
            print(f"实时快照已保存: {filepath}")
    finally:
        tq.close()

五、 方案总结与建议
格式统一：本方案定义了清晰、可扩展的CSV格式，适用于大多数量化分析场景。
来源兼容：字段设计参考了主流数据源，便于将从Tushare、AKShare、QMT、TdxQuant获取的数据统一到同一格式。
实践路径：
起步：先实现历史K线数据的定期下载和存储。
扩展：逐步加入实时快照、基本面数据，并建立自动化更新管道。
优化：随着数据量增长，可考虑将CSV迁移至数据库（如DuckDB、PostgreSQL）或列式存储（如Parquet），以提升查询性能。
关键点：无论使用哪种API，核心是将获取的原始数据映射到上述标准字段，确保整个项目数据格式的一致性。
此格式方案为您构建数据仓库奠定了坚实基础，后续可基于此进行数据清洗、因子计算和策略回测。
