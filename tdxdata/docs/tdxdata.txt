根据您的要求，我将制定一个通过API接口读取通达信历史行情和实时行情数据并写入CSV文件的方案。该方案主要基于知识库中提供的通达信TdxQuant官方接口。

方案概述
本方案使用通达信官方Python插件 tqcenter (TdxQuant)，通过外部调用方式连接通达信终端，获取历史K线数据与实时快照数据，并转换为规整的 pandas.DataFrame 格式，最终保存为CSV文件。

核心步骤与代码实现
1. 环境准备与外部调用配置
由于通达信插件安装在通达信目录下，需动态添加路径以实现外部环境（如PyCharm/VSCode）调用。

2. 数据清洗函数（将原始数据转换为DataFrame）
tq.get_market_data 返回的原始数据为字典嵌套Series格式，需转换为规整表格。

def raw_data_to_dataframe_complete(raw_data):
    """
    将tq.get_market_data返回的原始数据完整转换为DataFrame
    支持多股票、多字段的完整转换
    """
    if not raw_data:
        return pd.DataFrame()
    
    # 提取所有字段名（如Open, High, Low, Close, Volume）
    fields = list(raw_data.keys())
    # 提取所有股票代码
    stock_codes = list(raw_data[fields[0]].keys()) if fields else []
    
    df_list = []
    for stock_code in stock_codes:
        stock_data = {}
        for field in fields:
            # 提取该字段下该股票的数据序列
            series = raw_data[field].get(stock_code)
            if series is not None:
                stock_data[field] = series.values
            else:
                stock_data[field] = np.nan
        
        # 构建该股票的临时DataFrame
        stock_df = pd.DataFrame(stock_data)
        stock_df['股票代码'] = stock_code
        
        # 补充日期索引（假设索引为日期）
        if hasattr(series, 'index'):
            stock_df['日期'] = series.index
        
        df_list.append(stock_df)
    
    # 合并所有股票数据
    if df_list:
        final_df = pd.concat(df_list, ignore_index=True)
        # 调整列顺序：代码、日期、指标...
        col_order = ['股票代码', '日期'] + fields
        # 确保列存在
        final_df = final_df[[col for col in col_order if col in final_df.columns]]
        return final_df
    return pd.DataFrame()
3. 获取历史行情数据并保存为CSV
支持日线、分钟线等多种周期。

def fetch_and_save_history_data(stock_list, start_date, end_date, period='1d', csv_filename='history_data.csv'):
    """
    获取历史行情数据并保存为CSV文件
    :param stock_list: 股票代码列表，如 ["600519.SH", "000001.SZ"]
    :param start_date: 起始日期，格式 "20260105"
    :param end_date: 结束日期，格式 "20260120"
    :param period: K线周期，支持 '1d'(日线), '1m'(1分钟), '5m', '15m', '30m', '1h' 等
    :param csv_filename: 输出的CSV文件名
    """
    try:
        # 初始化与通达信的连接
        tq.initialize(__file__)
        print("成功与通达信建立连接！")
        
        # 刷新行情缓存，确保获取最新数据
        refresh_cache = tq.refresh_cache()
        print(f"刷新行情缓存结果: {refresh_cache}")
        
        # 获取原始数据
        raw_data = tq.get_market_data(
            stock_list=stock_list,
            start_time=start_date,
            end_time=end_date,
            period=period,
            dividend_type='front',  # 前复权。可选 'front'(前复权), 'back'(后复权), 'none'(不复权)
            field_list=['Open', 'High', 'Low', 'Close', 'Volume', 'Amount'],  # 需要的字段
            fill_data=True  # 填充缺失数据
        )
        
        # 转换为标准DataFrame
        df_standard = raw_data_to_dataframe_complete(raw_data)
        
        if not df_standard.empty:
            # 保存为CSV文件
            df_standard.to_csv(csv_filename, index=False, encoding='utf-8-sig')
            print(f"历史行情数据已保存至: {csv_filename}")
            print(f"数据形状: {df_standard.shape}")
            print(df_standard.head())
        else:
            print("未获取到有效数据。")
            
    except Exception as e:
        print(f"获取或处理数据时出错: {e}")
        import traceback
        traceback.print_exc()
    finally:
        # 清理资源，防止通达信卡死
        try:
            tq.close()
            print("通达信连接已清理")
        except:
            pass

4. 获取实时行情数据（快照）并保存为CSV
实时数据为当前时刻的盘口快照。

def fetch_and_save_realtime_snapshot(stock_list, csv_filename='realtime_snapshot.csv'):
    """
    获取实时行情快照数据并保存为CSV文件
    :param stock_list: 股票代码列表，如 ["600519.SH", "000001.SZ"]
    :param csv_filename: 输出的CSV文件名
    """
    try:
        tq.initialize(__file__)
        print("成功与通达信建立连接！")
        
        snapshot_list = []
        for code in stock_list:
            # 获取单只股票的实时快照
            market_snapshot = tq.get_market_snapshot(stock_code=code)
            if market_snapshot and 'ErrorId' in market_snapshot and market_snapshot['ErrorId'] == '0':
                # 添加股票代码
                market_snapshot['股票代码'] = code
                # 添加时间戳
                market_snapshot['更新时间'] = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
                snapshot_list.append(market_snapshot)
                print(f"已获取 {code} 实时快照")
            else:
                print(f"获取 {code} 实时快照失败或数据无效")
        
        if snapshot_list:
            # 转换为DataFrame并保存
            df_snapshot = pd.DataFrame(snapshot_list)
            df_snapshot.to_csv(csv_filename, index=False, encoding='utf-8-sig')
            print(f"实时行情快照已保存至: {csv_filename}")
            print(df_snapshot.head())
        else:
            print("未获取到任何有效的实时快照数据。")
            
    except Exception as e:
        print(f"获取实时数据时出错: {e}")
        import traceback
        traceback.print_exc()
    finally:
        try:
            tq.close()
            print("通达信连接已清理")
        except:
            pass

# 示例调用：获取多只股票实时快照
if __name__ == "__main__":
    fetch_and_save_realtime_snapshot(
        stock_list=["600519.SH", "000001.SZ", "000858.SZ"],
        csv_filename='realtime_snapshot_20260405.csv'
    )

关键注意事项

路径配置：tdx_install_path 必须修改为您的通达信软件实际安装路径。
通达信运行：执行脚本前，请确保通达信金融终端（64位量化版）已启动。
数据周期：
历史数据：period 支持 '1d' (日线)、'1m' (1分钟)、'5m'、'15m'、'30m'、'1h'、'1w'、'1mon' 等。
分钟数据：盘中可能无法下载，建议盘后获取。
实时数据：get_market_snapshot 返回的是包含五档行情等信息的快照，非Tick级流数据。
资源清理：务必在 finally 块中调用 tq.close()，防止通达信进程卡死。
数据字段：可根据需要调整 field_list，例如添加 'Amount' (成交额) 等。
